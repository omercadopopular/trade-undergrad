\section{Intro and recap}

\begin{frame}{So far\ldots}
\begin{wideitemize}
  \item \blue{Ricardian:} differences in technology induce comparative advantage; constant returns
  \item<2-> \blue{Specific-Factors:} specific factors $\rightarrow$ decreasing returns $\rightarrow$ distributional effects
  \item<3-> \blue{Heckscher–Ohlin:} same tech, different endowments $\rightarrow$ patterns of specialization
  \item<4-> \blue{Standard Trade Model:} trade happens due to differences $\rightarrow$ changes in terms of trade 
  \item<5-> \blue{In common}: we trade because we are different (in technologies, endowments, factors) \\
  \textcolor{gray}{(Florida is good at producing oranges, Georgia is good at producing peaches)}
  \item<6-> \blue{Next}: interindustry trade -- Krugman (time permitting) Melitz models\\
  \textcolor{gray}{(American cars are \textit{differentiated} from Euro cars; demand for both)}
\end{wideitemize}
\end{frame}

\begin{frame}{This class}
\begin{wideitemize}
    \item Shared patterns of all of these models: ``standard trade model''
    \item The Gravity Equation
    \item Empirical estimates of welfare gains
    \item Data Lab will go hands-on later.
\end{wideitemize}
\end{frame}

\begin{frame}{Key STM objects and notation}
\begin{wideitemize}

  \item Domestic \blue{relative supply:} $RS(p) \equiv \frac{Y_M(p)}{Y_A(p)}$

  \item Terms of Trade: $\text{ToT} = \frac{P_{\text{exports}}}{P_{\text{imports}}}$
  \begin{itemize}
      \item Do ToT of either country increase or decrease after they open up to trade in the STM?
  \end{itemize}
\end{wideitemize}

  \item \blue{Autarky:} $p^A$ at $RS(p^A)=RD(p^A)$

      RS(p) = \frac{Y_{H,M}}{Y_{H,A}}, \qquad RD(p) = \frac{Q_{H,M}}{Q_{H,A}}
  \end{equation*}
  \item<2-> \blue{With trade:} $p^w$ at $RS(p^w)=RD(p^w)$
  \begin{equation*}
      RS(p) = \frac{Y_{H,M}+ Y_{F,M}}{Y_{H,A} + Y_{F,A}}, \qquad RD(p) = \frac{Q_{H,M} + Q_{F,M}}{Q_{H,A} + Q_{F,A}}
  \end{equation*}
  \item<3-> \blue{Production after trade:} price $p^w$ pins production with slope $-p^w$
  \item<4-> \blue{Gains from trade:} higher affordable indifference curve if $p^w \neq p^A$
\end{wideitemize}
\end{frame}

\begin{frame}{From the Standard Trade Model to Trade Flows}
\small
\begin{itemize}
    \item \blue{Production side:} suppose each good $p$ in country $i$ requires $a_{i,p}$ units of labor per unit output \\
    \textcolor{gray}{(as in the Ricardian model)}

    \item<2-> Goods produced and sold in the domestic market are priced to their MC:
          \[
          P_{p,i} = a_{i,p} w_i \quad \text{(zero profits)}
          \]
    \item<3-> \blue{Demand side:} Cobb--Douglas preferences, as usual:
          \[
          U_i = Q_{i,M}^{\alpha_i} Q_{i,A}^{1-\alpha_i}, 
          \quad Q_{i,M} = \alpha_i \frac{w_iL_i}{P_{i,A}}.
          \]
          Each country spends a fixed share $\alpha_i$ of income on computers and $1-\alpha_i$ on roses.

\end{itemize}
\end{frame}


\begin{frame}{Iceberg Trade Costs}
\small
\begin{itemize}
    \item<1-> \blue{Iceberg trade cost}:
          to deliver one unit abroad, exporters must ship $\tau_{s\to d}\ge 1$ units.
    \item<2-> Delivered price in the destination:
          \[
          \underbrace{P_{sd,M}}_{\text{price at destination}} = \underbrace{\tau_{s\to d}}_{\text{trade cost} \ge1} \times \underbrace{P_{s,M}}_{\text{price at source}} = \tau_{s\to d} \times \, a_{M,s}\, w_s
          \]
\end{itemize}
\end{frame}


\begin{frame}{From Quantities to the Gravity Equation}
\small

\begin{wideitemize}
    \item Resulting import demand (quantities)
\[
Q_{M,s\to d} = 
\alpha_d \frac{w_d L_d}{\tau_{s\to d}\, a_{C,s}\, w_s} =  \underbrace{\frac{1}{ a_{M,s}\, w_s}}_{\text{source factors}} \times \underbrace{\alpha_d w_d L_d}_{\text{destination factors}} \times \underbrace{\frac{1}{\tau_{s\to d}}}_{\text{bilateral trade costs}}
\]
\textcolor{gray}{(Imports rise with importer income, and fall with trade and production costs.)}

\onslide<2->{
\begin{itemize}
    \item increasing in the importing country’s income ($w_d L_d$),
    \item decreasing in bilateral trade costs ($\tau_{s \to d}$),
    \item decreasing in the exporter’s unit labor requirement ($a_{p,s}$), and
    \item decreasing in the exporter’s wage ($w_s$), which represent production costs.
\end{itemize}
}

\item<3-> In general, exports from the $s$ to $d$ will satisfy:  

\[
\underbrace{X_{sd}}_{\substack{\text{trade flows}\\ \text{from s to d} }} \propto \frac{\underbrace{\omega_{s}}_{\text{source factors}} \times \underbrace{\gamma_{d}}_{\text{destination factors}}}{\underbrace{\tau_{sd}^{\theta}}_{\text{bilateral trade costs}}} 
\]

\end{wideitemize}


\end{frame}

\section{Gravity equation}

\begin{frame}{Gravity: Physics vs. Economics}
\begin{columns}[T]
\begin{column}{0.48\textwidth}
\blue{Newton's Law of Gravitation}
\[
F_{ij} \;=\; G\,\frac{m_i\,m_j}{r_{ij}^{\,2}}
\]
\begin{wideitemize}
  \item $m_i, m_j$: physical masses
  \item $r_{ij}$: distance between objects
  \item $G$: universal gravitational constant
\end{wideitemize}
\end{column}
\begin{column}{0.48\textwidth}
\onslide<2->{
\blue{Trade Gravity (reduced form)}
\[
X_{ij} \;=\; K \,\frac{Y_i\,Y_j}{d_{ij}^{\,\theta}}
\]
\begin{wideitemize}
  \item $Y_i, Y_j$: \emph{economic masses} (GDP/total expenditure)
  \item $d_{ij}$: bilateral distance (proxy for trade costs)
  \item $\theta>0$: trade‐cost elasticity (empirical)
  \item $K$: scaling constant / fixed effects
\end{wideitemize}
}
\end{column}
\end{columns}
\end{frame}

\begin{frame}{Measuring Trade Costs: \textit{Railroads of the Raj}}
\begin{itemize}
    \item \blue{Goal:} Quantify how India's 19th–20th century railroad expansion reduced trade costs (and raised welfare).
    \item<2-> \blue{Idea:} When trade is costly, identical goods sell for different prices across regions:
    \[
    p_i = p_j \times \tau_{ij}
    \quad \implies \quad 
    \ln(p_i/p_j) = \ln(\tau_{ij})
    \]
    \item<3-> \blue{Natural experiment}: expansion of India's railway system during colonial times.
    \item<4-> \blue{Salt as a reference good}: tradeable and homogeneous; limited local production.
    \item<5-> \blue{Intuition}: expansion of railroads reduce trade costs $\to$ prices of salt should converge.
\end{itemize}
\end{frame}

\begin{frame}{Measuring Trade Costs: \textit{Railroads of the Raj}}
\begin{itemize}
    \item Donaldson (2018) uses \blue{price differences} between district pairs as a \textit{revealed measure} of trade frictions.
     \item Empirical specification:
    \[
    \ln\!\left(\frac{p_{it}}{p_{jt}}\right)
    = \alpha + \beta\,\text{Effective Distance}_{ijt} + \gamma X_{ijt} + \varepsilon_{ijt}
    \]
    
    \item Effective distance:
    \begin{itemize}
        \item shortest available travel distance, given the transport network available in a given year — taking into account which transport modes exist (rail, river, road) and their relative cost-efficiency.
    \end{itemize}
    \item $\beta>0$  $\Rightarrow$  rail connection $\Rightarrow$ smaller effective distance $\Rightarrow$ smaller price gaps $\Rightarrow$ lower trade costs.
\end{itemize}
\end{frame}
